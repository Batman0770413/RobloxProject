local function getPlayerFactory()
	for _, factory in ipairs(factoriesFolder:GetChildren()) do
		if factory:GetAttribute("OwnerUserId") == player.UserId then
			return factory
		end
	end
	return nil
end

local function snap(value)
	return math.floor((value / GRID_SIZE) + 0.5) * GRID_SIZE
end

local function snapPosition(position)
	return Vector3.new(
		snap(position.X),
		position.Y,
		snap(position.Z)
	)
end

local function updateTextBoxes(pos, rot, size)
	transformTextBoxes.PosX.Text = string.format("%.1f", pos.X)
	transformTextBoxes.PosY.Text = string.format("%.1f", pos.Y)
	transformTextBoxes.PosZ.Text = string.format("%.1f", pos.Z)
	transformTextBoxes.RotX.Text = string.format("%.0f", rot.X)
	transformTextBoxes.RotY.Text = string.format("%.0f", rot.Y)
	transformTextBoxes.RotZ.Text = string.format("%.0f", rot.Z)
	transformTextBoxes.SizeX.Text = string.format("%.1f", size.X)
	transformTextBoxes.SizeY.Text = string.format("%.1f", size.Y)
	transformTextBoxes.SizeZ.Text = string.format("%.1f", size.Z)
end

local function readTextBoxes()
	local function num(text)
		local n = tonumber(text)
		return n or 0
	end

	local pos = Vector3.new(
		num(transformTextBoxes.PosX.Text),
		num(transformTextBoxes.PosY.Text),
		num(transformTextBoxes.PosZ.Text)
	)
	local rot = Vector3.new(
		num(transformTextBoxes.RotX.Text),
		num(transformTextBoxes.RotY.Text),
		num(transformTextBoxes.RotZ.Text)
	)
	local size = Vector3.new(
		math.clamp(num(transformTextBoxes.SizeX.Text), MIN_SIZE.X, MAX_SIZE.X),
		math.clamp(num(transformTextBoxes.SizeY.Text), MIN_SIZE.Y, MAX_SIZE.Y),
		math.clamp(num(transformTextBoxes.SizeZ.Text), MIN_SIZE.Z, MAX_SIZE.Z)
	)

	return pos, rot, size
end

local function showTransformPanel(block)
	if not block or not block:IsA("BasePart") then return end

	originalValues = {
		Position = block.Position,
		Rotation = block.Orientation,
		Size = block.Size
	}

	updateTextBoxes(block.Position, block.Orientation, block.Size)
	transformPanel.Visible = true
end

local function hideTransformPanel()
	transformPanel.Visible = false
end

local function applyTransform()
	if not selectedBlock or not selectedBlock:IsDescendantOf(workspace) then
		hideTransformPanel()
		return
	end

	local pos, rot, size = readTextBoxes()

	transformPartEvent:FireServer(selectedBlock, pos, rot, size)
	hideTransformPanel()
end

local function cancelTransform()
	if originalValues and selectedBlock then
		updateTextBoxes(originalValues.Position, originalValues.Rotation, originalValues.Size)
	end
	hideTransformPanel()
end

local function resetTransform()
	if originalValues then
		updateTextBoxes(originalValues.Position, originalValues.Rotation, originalValues.Size)
	end
end

local function quickMove(axis, direction)
	local pos, rot, size = readTextBoxes()
	if axis == "X" then
		pos = Vector3.new(pos.X + direction * GRID_SIZE, pos.Y, pos.Z)
	elseif axis == "Y" then
		pos = Vector3.new(pos.X, pos.Y + direction * GRID_SIZE, pos.Z)
	elseif axis == "Z" then
		pos = Vector3.new(pos.X, pos.Y, pos.Z + direction * GRID_SIZE)
	end
	updateTextBoxes(pos, rot, size)
end

local function quickRotate(axis)
	local pos, rot, size = readTextBoxes()
	if axis == "X" then
		rot = Vector3.new(rot.X + 90, rot.Y, rot.Z)
	elseif axis == "Y" then
		rot = Vector3.new(rot.X, rot.Y + 90, rot.Z)
	elseif axis == "Z" then
		rot = Vector3.new(rot.X, rot.Y, rot.Z + 90)
	end
	updateTextBoxes(pos, rot, size)
end

local function quickScale(direction)
	local pos, rot, size = readTextBoxes()
	local delta = direction * 1
	size = Vector3.new(
		math.clamp(size.X + delta, MIN_SIZE.X, MAX_SIZE.X),
		math.clamp(size.Y + delta, MIN_SIZE.Y, MAX_SIZE.Y),
		math.clamp(size.Z + delta, MIN_SIZE.Z, MAX_SIZE.Z)
	)
	updateTextBoxes(pos, rot, size)
end

local function getMouseRaycast()
	local factory = getPlayerFactory()
	if not factory then return nil end

	local mouseLocation = UserInputService:GetMouseLocation()
	local ray = camera:ViewportPointToRay(mouseLocation.X, mouseLocation.Y)

	local buildArea = factory:FindFirstChild("BuildArea")
	local playerBuilds = factory:FindFirstChild("PlayerBuilds")

	if not buildArea or not playerBuilds then return nil end

	local raycastParams = RaycastParams.new()
	raycastParams.FilterType = Enum.RaycastFilterType.Include
	raycastParams.FilterDescendantsInstances = { buildArea, playerBuilds }

	return workspace:Raycast(ray.Origin, ray.Direction * 1000, raycastParams)
end

local function getBuildPosition(result, buildArea)
	local target = result.Instance
	local normal = result.Normal
	local hitPosition = result.Position

	if target.Name == "BuildArea" then
		local topY = target.Position.Y + target.Size.Y / 2
		local blockCenterY = topY + BLOCK_SIZE.Y / 2
		return Vector3.new(snap(hitPosition.X), blockCenterY, snap(hitPosition.Z))
	end

	local position = hitPosition + Vector3.new(
		normal.X * BLOCK_SIZE.X / 2,
		normal.Y * BLOCK_SIZE.Y / 2,
		normal.Z * BLOCK_SIZE.Z / 2
	)

	local topY = buildArea.Position.Y + buildArea.Size.Y / 2
	local baseY = topY + BLOCK_SIZE.Y / 2
	local yOffset = position.Y - baseY
	local snappedY = baseY + math.floor(yOffset / BLOCK_SIZE.Y + 0.5) * BLOCK_SIZE.Y

	return Vector3.new(snap(position.X), snappedY, snap(position.Z))
end

RunService.RenderStepped:Connect(function()
	local factory = getPlayerFactory()

	if not factory then
		previewBlock.Transparency = 1
		highlight.Enabled = false
		positionLabel.Text = "لم يتم العثور على مصنعك"
		currentPosition = nil
		currentTarget = nil
		currentNormal = nil
		return
	end

	local result = getMouseRaycast()

	if selectedTool.Value == "Block" then
		highlight.Enabled = false

		if not result then
			previewBlock.Transparency = 1
			currentPosition = nil
			currentTarget = nil
			currentNormal = nil
			positionLabel.Text = "وجّه المؤشر إلى منطقة البناء"
			return
		end

		local buildArea = factory:FindFirstChild("BuildArea")
		if not buildArea then
			previewBlock.Transparency = 1
			return
		end

		local buildPosition = getBuildPosition(result, buildArea)
		currentPosition = buildPosition
		currentTarget = result.Instance
		currentNormal = result.Normal

		previewBlock.Position = buildPosition
		previewBlock.Transparency = 0.5

		positionLabel.Text = string.format("مكان البناء: X %.0f | Y %.0f | Z %.0f", buildPosition.X, buildPosition.Y, buildPosition.Z)

	elseif selectedTool.Value == "Select" then
		previewBlock.Transparency = 1

		local playerBuilds = factory:FindFirstChild("PlayerBuilds")

		if result and playerBuilds and result.Instance:IsDescendantOf(playerBuilds) then
			selectedBlock = result.Instance
			highlight.Adornee = selectedBlock
			highlight.Enabled = true

			local pos = selectedBlock.Position
			positionLabel.Text = string.format("البلوك المحدد: X %.0f | Y %.0f | Z %.0f", pos.X, pos.Y, pos.Z)
		else
			selectedBlock = nil
			highlight.Adornee = nil
			highlight.Enabled = false
			positionLabel.Text = "مرّر المؤشر فوق بلوكة من بنائك"
		end

	else
		previewBlock.Transparency = 1
		highlight.Enabled = false
		selectedBlock = nil
		positionLabel.Text = "اختر أداة"
	end
end)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if input.UserInputType ~= Enum.UserInputType.MouseButton1 then return end

	if selectedTool.Value == "Block" then
		if currentPosition then
			placePartEvent:FireServer("Block", currentPosition, currentTarget, currentNormal)
		end

	elseif selectedTool.Value == "Select" then
		local result = getMouseRaycast()
		if result then
			local factory = getPlayerFactory()
			local playerBuilds = factory and factory:FindFirstChild("PlayerBuilds")

			if playerBuilds and result.Instance:IsDescendantOf(playerBuilds) then
				-- النقرة الثانية على نفس البلوك تفتح TransformPanel
				if lastClickedBlock == result.Instance and selectedBlock == result.Instance then
					showTransformPanel(result.Instance)
				else
					selectedBlock = result.Instance
					lastClickedBlock = result.Instance
					highlight.Adornee = selectedBlock
					highlight.Enabled = true

					local position = selectedBlock.Position
					positionLabel.Text = string.format("تم تحديد البلوكة: X %.0f | Y %.0f | Z %.0f", position.X, position.Y, position.Z)
				end
			else
				lastClickedBlock = nil
			end
		end

	elseif selectedTool.Value == "Delete" then
		local result = getMouseRaycast()
		if result then
			deletePartEvent:FireServer(result.Instance)
		end
	end
end)
